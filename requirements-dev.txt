# this file, unless you are regenerating the locks:
#
#   python -m pip install --no-deps -r requirements/lock-py312-win_amd64.txt
#   python -m pip install --no-deps --no-build-isolation -e ./pob2-mcp -e ./pob2-optimizer

# First create/activate a clean venv and install pip, setuptools and wheel.
# Verify after installing: python scripts/lock_dependencies.py --check

#
#   py -3.10 scripts/lock_dependencies.py
#   py -3.12 scripts/lock_dependencies.py
#
